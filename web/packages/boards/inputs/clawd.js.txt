/* ClawdScene — ASCII / sub-pixel Clawd that lives on a canvas.
   Sprite: 18×5 sub-pixels (1:2 aspect), decoded from the three terminal lines
     ▐▛███▜▌ / ▝▜█████▛▘ / ▘▘ ▝▝
   Everything is drawn as colored cells + a textured glyph, like the whale. */
(function () {
  const PALS = {
    orange: { paper: '#faf9f5', body: '#d97757', tex: '#b85f3f', hi: '#e8926f', ink: '#141413', shadow: '#141413', spin: '#d97757', friends: ['#d97757', '#6a9bcc', '#788c5d', '#b4a0d6', '#c9a227'] },
    blue:   { paper: '#faf9f5', body: '#6a9bcc', tex: '#4f7fae', hi: '#8fb4da', ink: '#141413', shadow: '#141413', spin: '#6a9bcc', friends: ['#6a9bcc', '#d97757', '#788c5d', '#b4a0d6', '#c9a227'] },
    night:  { paper: '#141413', body: '#d97757', tex: '#f0a07c', hi: '#ffb996', ink: '#faf9f5', shadow: '#faf9f5', spin: '#d97757', friends: ['#d97757', '#6a9bcc', '#9db37c', '#c3b1e1', '#e3c35c'] }
  };
  const SPIN = ['·', '✢', '✳', '✶', '✻', '✽'];
  const TEX = ['▒', '▒', '░', '▓', '▒', '░', '▒', '▓'];
  const LEGS = [[4, 6, 11, 13], [3, 6, 11, 14], [4, 6, 11, 13], [5, 6, 11, 12]];
  const TUCK = [5, 6, 11, 12];

  function h3(a, b, c) { let h = (a * 374761393 + b * 668265263 + c * 2147483647) | 0; h = (h ^ (h >>> 13)) * 1274126177; return ((h ^ (h >>> 16)) >>> 0); }
  const clamp = (v, a, b) => v < a ? a : v > b ? b : v;
  const lerp = (a, b, t) => a + (b - a) * t;
  const ease = t => t < .5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
  const rnd = (a, b) => a + Math.random() * (b - a);

  function cells(pose) {
    const out = [];
    const push = (c, r, k) => out.push([c, r, k || 0]);
    const eyes = pose.blink ? [] : [[5 + pose.eyeDx, 1 + pose.eyeDy], [12 + pose.eyeDx, 1 + pose.eyeDy]];
    const isEye = (c, r) => eyes.some(e => e[0] === c && e[1] === r);
    const hd = pose.headDx || 0;
    for (let c = 3; c <= 14; c++) if (!isEye(c, 0)) push(c + hd, 0);
    for (let c = 3; c <= 14; c++) if (!isEye(c, 1)) push(c + hd, 1);
    for (let c = 1; c <= 16; c++) push(c, 2);
    for (let c = 3; c <= 14; c++) push(c, 3);
    if (pose.legs) for (const c of pose.legs) push(c, 4, 1);
    if (pose.claw === 1) { push(17, 1, 2); push(18, 0, 2); }
    if (pose.claw === 2) { push(17, 0, 2); push(18, -1, 2); }
    if (pose.clawL === 1) { push(0, 1, 2); push(-1, 0, 2); }
    if (pose.clawL === 2) { push(0, 0, 2); push(-1, -1, 2); }
    return out;
  }

  class ClawdScene {
    constructor(canvas, opts) {
      this.cv = canvas;
      this.ctx = canvas.getContext('2d');
      this.o = Object.assign({
        palette: 'orange', sizeFrac: .27, minCell: 5, maxCell: 40,
        floor: .57,
        friends: true, friendCount: 3, spinner: true, floorLine: true,
        texture: true, entrance: true, sleepAfter: 28, speed: 1,
        transparent: false, helmet: false, bubbles: false, onDone: null
      }, opts || {});
      this.t = 0; this.last = performance.now();
      this.W = 0; this.H = 0; this.dpr = 1; this.CW = 10;
      this.x = 0; this.dir = 1; this.hopY = 0; this.bob = 0;
      this.state = 'idle'; this.stateT = 0; this.stateDur = 1.5; this.stateData = {};
      this.eye = { dx: 0, dy: 0, tx: 0, ty: 0, hold: 0 };
      this.blink = { t: 0, next: rnd(2, 5) };
      this.legFrame = 0; this.stepPhase = 0;
      this.claw = 0; this.clawL = 0;
      this.spin = 0; this.spinOn = false;
      this.zs = []; this.bubs = []; this.lastInteract = 0;
      this.mouse = null; this.friends = [];
      this.texPhase = 0;
      this.onResize = () => this.resize();
      this.onMove = e => { const r = this.cv.getBoundingClientRect(); this.mouse = { x: e.clientX - r.left, y: e.clientY - r.top }; this.wake(); };
      this.onLeave = () => { this.mouse = null; };
      this.onClick = e => { const r = this.cv.getBoundingClientRect(); this.poke(e.clientX - r.left, e.clientY - r.top); };
      window.addEventListener('resize', this.onResize);
      canvas.addEventListener('pointermove', this.onMove);
      canvas.addEventListener('pointerleave', this.onLeave);
      canvas.addEventListener('pointerdown', this.onClick);
      this.resize();
      this.x = this.o.entrance ? -12 * this.CW : this.W * .5;
      if (this.o.entrance) this.go('walk', { to: this.W * .5 });
      this.seedFriends();
      this.raf = requestAnimationFrame(ts => this.frame(ts));
    }

    /* ---------- public ---------- */
    setOptions(o) {
      const sizeChanged = o.sizeFrac !== undefined && o.sizeFrac !== this.o.sizeFrac;
      Object.assign(this.o, o);
      if (sizeChanged) this.resize();
      if (o.friendCount !== undefined) this.seedFriends();
    }
    hop() { if (this.state !== 'hop') this.go('hop', { prev: this.state }); this.wake(); }
    wave() { this.go('wave'); this.wake(); }
    think(on) { this.spinOn = !!on; if (on) { this.go('think', { sticky: true }); this.wake(); } else if (this.state === 'think') this.go('idle'); }
    react() { this.spinOn = false; this.go('hop', { then: 'wave' }); this.wake(); }
    lookAt(x, y) { this.mouse = { x, y }; }
    poke(x, y) {
      this.wake();
      const half = 9 * this.CW, top = this.floorY() - 5 * this.CH - this.hopY;
      if (Math.abs(x - this.x) < half + this.CW && y > top - this.CH && y < this.floorY() + this.CH) this.react();
      else this.go('walk', { to: clamp(x, 10 * this.CW, this.W - 10 * this.CW) });
    }
    restart() { this.x = -12 * this.CW; this.go('walk', { to: this.W * .5 }); }
    pos() {
      const fy = this.floorY();
      return { x: this.x, y: fy - 2.5 * this.CH - this.hopY - this.bob, w: 18 * this.CW, h: 5 * this.CH, floor: fy, CW: this.CW, CH: this.CH, state: this.state, dir: this.dir };
    }
    active() { return this.x > -14 * this.CW && this.x < this.W + 14 * this.CW; }
    destroy() {
      cancelAnimationFrame(this.raf);
      window.removeEventListener('resize', this.onResize);
      this.cv.removeEventListener('pointermove', this.onMove);
      this.cv.removeEventListener('pointerleave', this.onLeave);
      this.cv.removeEventListener('pointerdown', this.onClick);
    }

    /* ---------- layout ---------- */
    resize() {
      const r = this.cv.getBoundingClientRect();
      this.W = Math.max(1, r.width); this.H = Math.max(1, r.height);
      this.dpr = Math.min(2, window.devicePixelRatio || 1);
      this.cv.width = Math.round(this.W * this.dpr); this.cv.height = Math.round(this.H * this.dpr);
      this.CW = clamp(Math.round(this.W * this.o.sizeFrac / 18), this.o.minCell, this.o.maxCell);
      this.CH = this.CW * 2;
      this.x = clamp(this.x, -14 * this.CW, this.W + 14 * this.CW);
    }
    floorY() { return Math.round(this.H * this.o.floor); }
    get CH_() { return this.CH; }
    pal() { return PALS[this.o.palette] || PALS.orange; }
    wake() { this.lastInteract = this.t; if (this.state === 'sleep') this.go('idle'); }

    /* ---------- behaviour ---------- */
    go(state, data) {
      this.state = state; this.stateT = 0; this.stateData = data || {};
      switch (state) {
        case 'idle': this.stateDur = rnd(1.8, 4.2); break;
        case 'walk': this.stateDur = 99; this.dir = (this.stateData.to >= this.x) ? 1 : -1; break;
        case 'hop': this.stateDur = .62; break;
        case 'wave': this.stateDur = 1.8; break;
        case 'think': this.stateDur = this.stateData.sticky ? 99 : rnd(2.4, 3.6); this.spin = 0; break;
        case 'sleep': this.stateDur = 99; break;
        case 'look': this.stateDur = 1.2; break;
      }
    }
    plan() {
      const idle = this.t - this.lastInteract;
      if (idle > this.o.sleepAfter && this.state === 'idle') return this.go('sleep');
      const r = Math.random();
      if (this.state === 'walk') return this.go(r < .35 ? 'think' : r < .55 ? 'wave' : 'idle');
      if (this.state === 'idle') {
        if (r < .45) return this.go('walk', { to: rnd(10 * this.CW, this.W - 10 * this.CW) });
        if (r < .62) return this.go('think');
        if (r < .74) return this.go('hop');
        if (r < .86) return this.go('wave');
        return this.go('look');
      }
      return this.go('idle');
    }
    step(dt) {
      this.stateT += dt;
      const s = this.state, d = this.stateData, CW = this.CW;
      let tx = 0, ty = 0;
      if (this.mouse) {
        const dx = this.mouse.x - this.x, dy = this.mouse.y - (this.floorY() - 3 * this.CH);
        tx = Math.abs(dx) > 3 * CW ? Math.sign(dx) : 0;
        ty = dy < -3 * this.CH ? -1 : 0;
      }
      if (s === 'walk') tx = this.dir;
      if (s === 'look') tx = d.dir || (d.dir = Math.random() < .5 ? -1 : 1);
      if (s === 'think') ty = -1, tx = this.stateT % 1.6 < .8 ? 1 : -1;
      if (s === 'sleep') tx = 0, ty = 0;
      this.eye.dx = tx; this.eye.dy = ty;

      this.blink.t += dt;
      if (this.blink.t > this.blink.next) { this.blink.t = 0; this.blink.next = rnd(2.2, 5.5); this.blink.on = .13; }
      if (this.blink.on > 0) this.blink.on -= dt;

      this.hopY = 0; this.claw = 0; this.clawL = 0; this.legs = LEGS[0];
      const sp = this.o.speed;
      if (s === 'walk') {
        const v = 7.5 * CW * sp;
        this.x += this.dir * v * dt;
        this.stepPhase += dt * 7 * sp;
        this.legFrame = Math.floor(this.stepPhase) % 4;
        this.legs = LEGS[this.legFrame];
        this.bob = Math.abs(Math.sin(this.stepPhase * Math.PI)) * this.CH * .22;
        if ((this.dir > 0 && this.x >= d.to) || (this.dir < 0 && this.x <= d.to)) { this.x = d.to; this.plan(); }
      } else {
        this.bob = (Math.sin(this.t * 2 * Math.PI / 2.4) + 1) * .5 * this.CH * (s === 'sleep' ? .1 : .14);
        this.legFrame = 0;
      }
      if (s === 'hop') {
        const p = this.stateT / this.stateDur;
        this.hopY = Math.sin(Math.PI * clamp(p, 0, 1)) * this.CH * 1.6;
        if (p > .15 && p < .85) this.legs = TUCK;
        this.bob = 0;
      }
      this.headDx = 0;
      if (s === 'wave') {
        const f = Math.floor(this.stateT * 6) % 2;
        if (this.stateT < 1.4) { this.headDx = f ? 1 : -1; this.legs = LEGS[f ? 1 : 3]; }
      }
      if (s === 'think') { this.spin += dt * 9; }
      if (s === 'sleep') {
        if (Math.random() < dt * .9) this.zs.push({ x: this.x + 9 * CW, y: this.floorY() - 5.5 * this.CH, t: 0, dur: rnd(1.8, 2.6), s: rnd(.6, 1) });
      }
      for (const z of this.zs) z.t += dt;
      this.zs = this.zs.filter(z => z.t < z.dur);

      // bubbles (helmet mode)
      if (this.o.bubbles && this.active()) {
        if (Math.random() < dt * 1.4) {
          const fy = this.floorY(), CH = this.CH;
          this.bubs.push({ x: this.x + rnd(-2, 4) * CW, y: fy - 5 * CH - this.hopY - this.bob - CH, t: 0, dur: rnd(1.6, 2.8), s: rnd(.4, 1.1), dx: rnd(-.4, .4) });
        }
      }
      for (const b of this.bubs) b.t += dt;
      this.bubs = this.bubs.filter(b => b.t < b.dur);

      if (this.stateT > this.stateDur && !(s === 'think' && d.sticky) && s !== 'sleep') {
        if (s === 'hop' && d.then) this.go(d.then);
        else this.plan();
      }
      if (s === 'sleep' && this.t - this.lastInteract < this.o.sleepAfter) this.go('idle');
      this.texPhase += dt * .8;

      if (this.o.friends) for (const f of this.friends) {
        f.x += f.dir * f.v * CW * dt; f.phase += dt * f.v * 1.3;
        if (f.dir > 0 && f.x > this.W + 20 * f.cw) { f.x = -20 * f.cw; }
        if (f.dir < 0 && f.x < -20 * f.cw) { f.x = this.W + 20 * f.cw; }
        f.blink -= dt; if (f.blink < -.15) f.blink = rnd(2, 6);
      }
    }
    seedFriends() {
      const n = this.o.friendCount, pal = this.pal();
      this.friends = [];
      for (let i = 0; i < n; i++) {
        const dir = Math.random() < .5 ? 1 : -1;
        this.friends.push({ x: rnd(0, this.W), dir, v: rnd(4.5, 7.5), phase: rnd(0, 10), cw: 0, col: pal.friends[i % pal.friends.length], lane: i % 3, blink: rnd(1, 5) });
      }
    }

    /* ---------- render ---------- */
    frame(ts) {
      const dt = Math.min(.05, (ts - this.last) / 1000); this.last = ts; this.t += dt;
      const r = this.cv.getBoundingClientRect();
      if (Math.abs(r.width - this.W) > 1 || Math.abs(r.height - this.H) > 1) this.resize();
      this.step(dt);
      this.draw();
      this.raf = requestAnimationFrame(t => this.frame(t));
    }
    drawSprite(ox, oy, cw, pose, col, texCol, alpha, texture) {
      const ctx = this.ctx, ch = cw * 2;
      ctx.globalAlpha = alpha;
      ctx.fillStyle = col;
      const cs = cells(pose);
      for (const [c, r] of cs) ctx.fillRect(Math.round(ox + c * cw), Math.round(oy + r * ch), cw, ch);
      if (texture && cw >= 7) {
        ctx.fillStyle = texCol; ctx.globalAlpha = alpha * .45;
        ctx.font = `${Math.round(ch * .92)}px "Geist Mono", "DejaVu Sans Mono", Menlo, Consolas, monospace`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        const ph = Math.floor(this.texPhase);
        for (const [c, r, k] of cs) {
          const hv = h3(c + 20, r + 20, ph + (k === 1 ? 3 : 0));
          const g = TEX[hv % TEX.length];
          ctx.fillText(g, Math.round(ox + c * cw) + cw / 2, Math.round(oy + r * ch) + ch / 2 + 1);
        }
      }
      ctx.globalAlpha = 1;
    }
    draw() {
      const ctx = this.ctx, pal = this.pal(), CW = this.CW, CH = this.CH;
      ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0);
      if (this.o.transparent) { ctx.clearRect(0, 0, this.W, this.H); }
      else { ctx.fillStyle = pal.paper; ctx.fillRect(0, 0, this.W, this.H); }
      const fy = this.floorY();

      if (this.o.floorLine && !this.o.transparent) {
        ctx.fillStyle = pal.ink; ctx.globalAlpha = .18;
        ctx.font = `${Math.round(Math.max(9, CW * .9))}px "Geist Mono", "DejaVu Sans Mono", Menlo, Consolas, monospace`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        const step = CW * 2;
        for (let x = step / 2; x < this.W; x += step) ctx.fillText('·', x, fy + CH * .45);
        ctx.globalAlpha = 1;
      }

      if (this.o.friends) for (const f of this.friends) {
        f.cw = Math.max(3, Math.round(CW * .22));
        const fch = f.cw * 2;
        const legs = LEGS[Math.floor(f.phase) % 4];
        const bobF = Math.abs(Math.sin(f.phase * Math.PI)) * fch * .22;
        this.drawSprite(f.x - 9 * f.cw, fy - 5 * fch - bobF, f.cw, { eyeDx: f.dir, eyeDy: 0, blink: f.blink < 0, legs }, f.col, f.col, .85, false);
      }

      // shadow on ground (skip in transparent+helmet mode — underwater has no ground shadow)
      if (!this.o.helmet) {
        const lift = this.hopY / (CH * 1.6);
        ctx.fillStyle = pal.shadow; ctx.globalAlpha = .22 * (1 - lift * .8);
        ctx.font = `${Math.round(CH * .8)}px "Geist Mono", "DejaVu Sans Mono", Menlo, Consolas, monospace`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        const shw = Math.round(7 - lift * 3);
        for (let i = -shw; i <= shw; i++) ctx.fillText(Math.abs(i) >= shw - 1 ? '·' : '-', Math.round(this.x + i * CW + CW * .5), fy + CH * .45);
        ctx.globalAlpha = 1;
      }

      // Clawd
      const pose = { headDx: this.headDx, eyeDx: this.eye.dx, eyeDy: this.eye.dy, blink: this.blink.on > 0 || this.state === 'sleep', legs: this.legs, claw: this.claw, clawL: this.clawL };
      const ox = this.x - 9 * CW, oy = fy - 5 * CH - this.hopY - this.bob;
      this.drawSprite(ox, oy, CW, pose, pal.body, pal.tex, 1, this.o.texture);

      // diving helmet (glass dome around head)
      if (this.o.helmet) {
        const hx = this.x, hy = oy + CH;
        const hr = CW * 5.5;
        // glass dome
        ctx.beginPath();
        ctx.arc(hx, hy, hr, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(140,200,220,0.08)';
        ctx.fill();
        ctx.strokeStyle = 'rgba(140,200,220,0.3)';
        ctx.lineWidth = Math.max(0.5, CW * 0.07);
        ctx.stroke();
        // highlight glint
        ctx.beginPath();
        ctx.arc(hx - hr * .35, hy - hr * .3, hr * .12, 0, Math.PI * 2);
        ctx.fillStyle = 'rgba(255,255,255,0.18)';
        ctx.fill();
        // rim at bottom
        ctx.beginPath();
        ctx.ellipse(hx, oy + CH * 2.5, CW * 5, CW * 1.2, 0, 0, Math.PI * 2);
        ctx.strokeStyle = 'rgba(140,200,220,0.2)';
        ctx.lineWidth = Math.max(0.5, CW * 0.1);
        ctx.stroke();
      }

      // bubbles (rising from helmet)
      if (this.bubs.length) {
        for (const b of this.bubs) {
          const p = b.t / b.dur;
          const bx = b.x + p * CW * 3 * b.dx + Math.sin(p * 7) * CW * .4;
          const by = b.y - p * CH * 5;
          const br = CW * (.25 + .2 * b.s) * (1 - p * .4);
          ctx.globalAlpha = (1 - p * p) * .35;
          ctx.beginPath();
          ctx.arc(bx, by, Math.max(0.5, br), 0, Math.PI * 2);
          ctx.strokeStyle = 'rgba(140,200,220,0.5)';
          ctx.lineWidth = 0.5;
          ctx.stroke();
          ctx.beginPath();
          ctx.arc(bx - br * .3, by - br * .3, Math.max(0.3, br * .18), 0, Math.PI * 2);
          ctx.fillStyle = 'rgba(255,255,255,0.2)';
          ctx.fill();
        }
        ctx.globalAlpha = 1;
      }

      // spinner
      if ((this.state === 'think' || this.spinOn) && this.o.spinner) {
        ctx.fillStyle = pal.spin; ctx.globalAlpha = .95;
        ctx.font = `${Math.round(CH * 1.35)}px "Geist Mono", "DejaVu Sans Mono", Menlo, Consolas, monospace`;
        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        const idx = Math.floor(this.spin) % SPIN.length;
        ctx.fillText(SPIN[idx], Math.round(this.x + 9 * CW + CW * 1.2), Math.round(oy - CH * .6));
        ctx.globalAlpha = 1;
      }
      // z's
      if (this.zs.length) {
        ctx.fillStyle = pal.ink; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        for (const z of this.zs) {
          const p = z.t / z.dur;
          ctx.globalAlpha = (1 - p) * .55;
          ctx.font = `${Math.round(CH * (.55 + .5 * p) * z.s)}px "Geist Mono", "DejaVu Sans Mono", Menlo, Consolas, monospace`;
          ctx.fillText('z', Math.round(z.x + p * CW * 3 + Math.sin(p * 6) * CW * .6), Math.round(z.y - p * CH * 2.2));
        }
        ctx.globalAlpha = 1;
      }
    }
  }
  window.ClawdScene = ClawdScene;
})();
