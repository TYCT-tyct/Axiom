//
// The sea and the sharks are Axiom's own; the whale (deepseek) and Clawd (claude) are guests. A guest wears the sea's
// material (grey tiles and glyphs, like the sharks and the wordmark) and shows its own colours only when it is found.
//
// 1. The dive. "claude" in the hero's family strip, clicked or tapped (or typed on a keyboard), brings a jellyfish
//    to the right edge with one Clawd hanging from it; every click after that sends one more out of the word to hang
//    on below the last, up to the whole team (three on a computer: a diving helmet, a scallop, a pearl; two on a
//    phone). A dashed line unrolls down the right edge: it is the depth of the page, and the jellyfish sinks along it
//    as you read, at its own pace (it falls behind when you scroll fast, catches up, and stays nearby when you stop),
//    greying back into the sea after a while. Tapping the team takes you down. The claude rows of the models table
//    wink as it passes. When the sea floor comes into view they let go: the jellyfish drifts up and away, each one
//    jumps into the sand behind its own letter (m, o, x), looks out and waves, and the letters A, x, i, o and m turn into
//    Clawds one after another.
// 2. The wordmark. A few Clawds live in the sand behind the letters. The first time the floor comes into view, the one behind
//    the m looks out in colour and waves. Now and then one looks over the top of its letter,
//    grey; pointing at a letter makes the one behind it look out in colour and wave; clicking (tapping) any letter, the i too,
//    breaks its tiles apart into a Clawd and back (pointing at the i makes its dot hop). The letters always stay in front: whatever hides behind them is cut out
//    of them, and a letter that turns into a Clawd is its own tiles.
//
// One director: Clawd never shares the water with the whale (it waits for it to leave, and the whale waits for it).

// depth line, since on the phone the content runs almost to the edge); the wordmark part is drawn
// by the seabed's canvas, which hands its letters over every frame (canvas.art.__fx: skip, dotDy, pre, post, fast).

      for (const a of out) drawMat(ctx, a.k, a.CW, Math.round(a.x - 9 * a.CW), Math.round(a.y - 5 * a.CW * 2), pose(a.p, { edx: a.claw === 'wave' ? 0 : a.p.edx }), pal, a.rev, a.claw ? claws(a.claw, Math.floor(a.p.t * 6) % 2) : [], 1);
      ctx.restore();
    }
    post(ctx, info) {
      const b = this.bed, pal = this.pal(), CW = info.CW, CH = info.CH, t = info.t;
      for (const key of Object.keys(b.morphs)) {
        const M = b.morphs[key], u = t - M.t0, F = 0.8, A = F + 0.45, B = A + M.life, Cc = B + 0.4, E = Cc + F;
        if (u > E) { delete b.morphs[key]; this.fx.skip.delete(M.li); continue; }
        tickPose(M.p, 1 / 30);
        ctx.font = `${info.GF}px ${FONT}`; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        if (u < F || u > Cc) {
          const p = u < F ? u / F : 1 - (u - Cc) / F;
          M.parts.forEach((q, j) => {
            const e = ss(q.dl, q.dl + 0.64, p), x = q.s.x + (q.d.x - q.s.x) * e, y = q.s.y + (q.d.y - q.s.y) * e - Math.sin(Math.PI * e) * q.arc, a = q.dup ? 1 - e : 1; if (a <= 0.02) return;
            const w = CW - 1 + (q.d.w - CW) * e, hh = CH - 1 + (q.d.h - CH) * e;
            ctx.globalAlpha = a * (0.26 + 0.52 * q.s.m); ctx.fillStyle = info.tile; ctx.fillRect(x - w / 2, y - hh / 2, w, hh);
            ctx.globalAlpha = a * (0.5 + 0.5 * q.s.m); ctx.fillStyle = info.ink;
            if (w >= 5) ctx.fillText(SOLID[h3(j, M.li, Math.floor(t * 4)) % SOLID.length], x, y + 0.5); else { ctx.globalAlpha *= 0.3; ctx.fillRect(x - w / 2, y - hh * 0.2, w, hh * 0.4); }
          });
          continue;
        }
        const Rv = ss(F, A, u) * (1 - ss(B, Cc, u)), hp = u > A && u < A + 0.5 ? Math.sin((u - A) / 0.5 * Math.PI) : 0, dy = -Math.round(hp * 2) * M.CW;
        M.dst.forEach((q, j) => {
          const r = clamp(Rv * 1.6 - (q.r + 6) / 11 * 0.6, 0, 1); if (r >= 1) return; const a = 1 - r, y = q.y + dy;
          ctx.globalAlpha = a * 0.78; ctx.fillStyle = info.tile; ctx.fillRect(q.x - q.w / 2 + 0.5, y - q.h / 2 + 0.5, q.w - 1, q.h - 1);
          ctx.globalAlpha = a; ctx.fillStyle = info.ink;
          if (q.w >= 5) ctx.fillText(SOLID[h3(j, M.li, Math.floor(t * 0.5)) % SOLID.length], q.x, y + 0.5); else if (h3(j, 2, 5) % 2) ctx.fillRect(q.x - q.w / 2, y - q.h * 0.2, q.w, q.h * 0.4);
        });
        // only the colour here: the material above already stands where the colour has not arrived yet
        const s = pose(M.p, { edx: hp > 0 ? 0 : M.p.edx }), cl = u > A + 0.4 && u < B - 0.1 ? claws('wave', Math.floor(t * 6) % 2) : hp > 0 ? claws('cheer', 1) : [];
        const f = OUT[M.k](s), ph = Math.floor(s.t * 0.8), cells = f.body.concat(cl.map((c) => ({ c: c[0], r: c[1], f: C.body })), f.post || []), MCH = M.CW * 2;
        ctx.font = Math.round(MCH * 0.86) + 'px ' + FONT;
        for (const q of cells) { const r = clamp(Rv * 1.6 - (q.r + 6) / 11 * 0.6, 0, 1); if (r > 0) cell(ctx, M.ox, M.oy + dy, M.CW, MCH, q, ph, pal, r); }
        ctx.globalAlpha = 1;
      }
    }

    // ---- the dive's own layer
    frame(dt) {
      // pointing at a letter for a moment makes the one behind it look out
      if (this.hl != null && !this.hlDone && performance.now() - this.hlAt > 260) { this.hlDone = true; if (this.hl === 2) this.hop(); else this.peek(this.hl, true); }
      this.floorSync();
      this.hint();
      // the first time the floor comes into view, the one behind the m looks out in colour and waves: someone lives here
      if (!this.greeted) {
        if (this.bed.info && this.floorInView() && !this.busy() && !this.whaleOut()) { const now = performance.now(); this.greetAt = this.greetAt || now + 1200; if (now > this.greetAt) { this.greeted = true; this.peek(4, true); } }
        else this.greetAt = 0;
      }
      if (this.pend && !this.whaleOut()) { const p = this.pend; this.pend = null; if (performance.now() - p.at < 15000) this.drop(p.anchor); }
      this.stepF(dt);
      // the team is drawn above the page while it is out (on the phone the content runs almost to the edge)
      const up = !!(this.F && this.F.st !== 'gone');
      if (up !== this.up) { this.up = up; this.cv.style.zIndex = up ? '2' : ''; }
      this.bubs = this.bubs.filter((b) => (b.age += dt) < b.dur);
      const F = this.F, live = (F && F.st !== 'gone') || this.bubs.length;
      if (!live && !this.drew) return;
      const ctx = this.ctx, pal = this.pal(); ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0); ctx.clearRect(0, 0, this.W, this.H); this.drew = !!live;
      if (!live) return;
      if (F && F.st !== 'gone') {
        // the depth line: the part above the jellyfish is where you have been; the team covers its own stretch
        const x = Math.round(this.trackX()) - 1, top = this.trackTop(), bot = this.H - 24, ride = F.st === 'ride', fill = ride ? F.jy : bot, step = 7;
        const c0 = F.jy - 4, c1 = F.jy + this.chainH(Math.max(1, F.m.filter((m) => m.hang).length)) + 4;
        for (let y = top; y < bot; y += step) {
          const q = (y - top) / (bot - top); if (q > F.trk) break; if (q < F.out * 1.05) continue; if (ride && y + step > c0 && y < c1) continue;
          const done = y < fill; ctx.globalAlpha = (done ? 0.5 : 0.22) * (1 - F.out); ctx.fillStyle = done ? pal.line : pal.tile; ctx.fillRect(x, y, 2, step - 3);
        }
      }
      ctx.font = '12px ' + FONT; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillStyle = pal.glass;
      for (const b of this.bubs) { const q = b.age / b.dur; ctx.globalAlpha = (1 - q) * 0.6; ctx.fillText(q < 0.5 ? 'o' : '°', b.x + Math.sin(q * 6 + b.sd) * 3, b.y - q * 40); }
      if (F && F.st !== 'gone') {
        const S = this.S, CH = S * 2, ga = F.st === 'leave' ? 1 - Math.min(1, F.u / 0.9) : 1, fast = F.st === 'ride' && Math.abs(F.vy) > 60, kick = F.cheer > 0 || fast;
        // the jellyfish: drawn as if it were the leader's hat, so the strings meet the leader's claws
        if (F.st === 'dive') { const B = F.bell, a = 1 - clamp(B.t / 2.6, 0, 1); if (a > 0) drawCells(ctx, bell(Math.sin(B.t * 4) > 0), S, Math.round(B.x - 9 * S + Math.sin(B.t * 1.4) * S * 2 - B.t * S * 5), Math.round(B.y + 6 * CH), Math.floor(B.t * 0.8), pal, B.rv, a); }
        else drawCells(ctx, bell(Math.sin(F.u * 2.6) > 0), S, Math.round(F.jx - 9 * S), Math.round(F.jy + this.bob() + 6 * CH), Math.floor(F.u * 0.8), pal, F.jr, ga * F.jin);
        for (let i = F.m.length - 1; i >= 0; i--) {
          const m = F.m[i]; if (m.done) continue;
          const wf = Math.floor(m.p.t * 6) % 2, cl = m.hop > 0.1 ? claws('cheer', 1) : F.st === 'dive' ? claws('cheer', wf) : F.cheer > 0 && i === F.m.length - 1 ? claws('wave', wf) : claws('hold');
          const s = pose(m.p, { legs: kick ? LEGS[(Math.floor(m.p.t * 8) + i) % 2 ? 1 : 3] : LEGS[0], edx: -1 });
          drawMat(ctx, m.k, S, Math.round(m.x - 9 * S), Math.round(m.y - 5 * CH), s, pal, m.rev, cl, ga);
        }
      }
      ctx.globalAlpha = 1;
    }
    destroy() {
      this.dead = true; cancelAnimationFrame(this.raf); if (this.ro) this.ro.disconnect();
      for (const [el, ev, fn, o] of this.on) el.removeEventListener(ev, fn, o);
      if (this.art && this.art.__fx === this.fx) delete this.art.__fx;
      const fe = this.famEl(); if (fe) fe.classList.remove('cw-lit');
      const w = this.eng && this.eng.world; if (w) { w.floor = null; w.floorTop = null; }
      if (this.clipOn && this.o.egg && this.o.egg.cv) this.o.egg.cv.style.clipPath = '';
    }
  }
  window.AxiomClawdCast = AxiomClawdCast;
})();
